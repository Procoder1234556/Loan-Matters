## CONTEXT (read before every phase)

You are building **LoanMatters** — an AI-powered education loan comparison platform for Indian STEM students planning to study abroad (US, UK, Canada, Germany, Australia). The live site is at `https://loan-matters.vercel.app/`. Stack: **Next.js 14 (App Router) + Tailwind CSS** on the frontend, **Node.js/Express** for the backend API, **PostgreSQL via Supabase** for data, and a **separate PHP 8+ admin panel** for blog management.

Core brand values:
- We never sell user data. We never spam. We just compare.
- Faster, cleaner, more honest than BankBazaar or Paisabazaar.
- Niche: education loans for abroad studies — go deep, not broad.

Stack constraints:
- Frontend: Next.js 14, App Router, Tailwind CSS, TypeScript
- Backend API: Node.js + Express (deployed on Railway)
- Database: Supabase (PostgreSQL)
- Admin panel: PHP 8 + MySQL (deployed on Hostinger or any VPS)
- Rate updates: Vercel Cron Jobs (every 6 hours)
- No `.env` file to be created or overwritten — use `.env.example` only

```typescript
// app/api/loans/route.ts
import { NextResponse } from 'next/server'
import { createClient } from '@supabase/supabase-js'

export const revalidate = 21600

export async function GET(request: Request) {
  const { searchParams } = new URL(request.url)
  const country = searchParams.get('country')
  const collateral = searchParams.get('collateral') // 'yes' | 'no' | 'any'
  const maxAmount = searchParams.get('max_amount_lakhs')

  const supabase = createClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!
  )

  let query = supabase
    .from('loan_products')
    .select(`*, lenders(name, slug, type, logo_url)`)
    .order('interest_rate_min', { ascending: true })

  if (country) {
    query = query.contains('countries_covered', [country.toUpperCase()])
  }
  if (collateral === 'no') {
    query = query.not('collateral_required_above_lakhs', 'is', null)
  }

  const { data, error } = await query

  if (error) return NextResponse.json({ error: 'Failed to fetch' }, { status: 500 })

  return NextResponse.json({
    data,
    meta: {
      total: data?.length ?? 0,
      last_updated: new Date().toISOString(),
      source: 'LoanMatters — loanmatters.in',
      disclaimer: 'Rates are indicative and sourced from lender websites. Verify with the lender before applying.'
    }
  }, {
    headers: {
      'Cache-Control': 'public, s-maxage=21600, stale-while-revalidate=86400',
      'Access-Control-Allow-Origin': '*'
    }
  })
}
```

```typescript
'use client'
import { useState, useMemo } from 'react'

type Filters = {
  country: string
  collateral: 'any' | 'yes' | 'no'
  maxAmountLakhs: number
  lenderType: 'all' | 'psu_bank' | 'private_bank' | 'nbfc'
  sortBy: 'rate' | 'amount' | 'verified'
}

export default function LoanCompareTable({ loans }: { loans: LoanProduct[] }) {
  const [filters, setFilters] = useState<Filters>({
    country: 'any',
    collateral: 'any',
    maxAmountLakhs: 150,
    lenderType: 'all',
    sortBy: 'rate'
  })
  const [selectedIds, setSelectedIds] = useState<string[]>([]) // for side-by-side compare

  const filtered = useMemo(() => {
    return loans
      .filter(l => {
        if (filters.country !== 'any' && !l.countries_covered?.includes(filters.country)) return false
        if (filters.lenderType !== 'all' && l.lenders?.type !== filters.lenderType) return false
        if (filters.collateral === 'no' && l.collateral_required_above_lakhs === null) return false
        return true
      })
      .sort((a, b) => {
        if (filters.sortBy === 'rate') return (a.interest_rate_min ?? 99) - (b.interest_rate_min ?? 99)
        if (filters.sortBy === 'amount') return (b.max_amount_abroad_lakhs ?? 0) - (a.max_amount_abroad_lakhs ?? 0)
        return 0
      })
  }, [loans, filters])

  return (
    <div className="grid grid-cols-1 lg:grid-cols-[280px_1fr] gap-6">
      <FilterSidebar filters={filters} onChange={setFilters} />
      <div className="space-y-4">
        <ResultsHeader count={filtered.length} sortBy={filters.sortBy} onSort={(s) => setFilters(f => ({...f, sortBy: s}))} />
        {filtered.map(loan => (
          <LoanCard
            key={loan.id}
            loan={loan}
            selected={selectedIds.includes(loan.id)}
            onSelect={(id) => setSelectedIds(prev =>
              prev.includes(id) ? prev.filter(x => x !== id) : prev.length < 3 ? [...prev, id] : prev
            )}
          />
        ))}
        {selectedIds.length >= 2 && (
          <StickyCompareBar
            selectedLoans={loans.filter(l => selectedIds.includes(l.id))}
            onClear={() => setSelectedIds([])}
          />
        )}
      </div>
    </div>
  )
}
```

```typescript
  // For each lender, fetch their official rate page and parse

  const results = await Promise.allSettled([
```

```php
session_start();

define('ADMIN_PASSWORD_HASH', '$2y$12$YOUR_BCRYPT_HASH_HERE'); // generate with password_hash()
define('SESSION_TIMEOUT', 3600); // 1 hour

function requireAuth(): void {
    if (!isset($_SESSION['admin_authenticated']) || !$_SESSION['admin_authenticated']) {
        header('Location: /admin/login.php');
        exit;
    }
    if (isset($_SESSION['last_activity']) && (time() - $_SESSION['last_activity']) > SESSION_TIMEOUT) {
        session_unset();
        session_destroy();
        header('Location: /admin/login.php?reason=timeout');
        exit;
    }
    $_SESSION['last_activity'] = time();
}
```
