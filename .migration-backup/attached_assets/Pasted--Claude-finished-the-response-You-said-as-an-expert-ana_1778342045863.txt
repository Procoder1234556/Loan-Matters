  const { searchParams } = new URL(request.url)
  const country = searchParams.get('country')
  const collateral = searchParams.get('collateral') // 'yes' | 'no' | 'any'
  const maxAmount = searchParams.get('max_amount_lakhs')

  const supabase = createClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!
  )

  let query = supabase
    .from('loan_products')
    .select(`*, lenders(name, slug, type, logo_url)`)
    .order('interest_rate_min', { ascending: true })

  if (country) {
    query = query.contains('countries_covered', [country.toUpperCase()])
  }
  if (collateral === 'no') {
    query = query.not('collateral_required_above_lakhs', 'is', null)
  }

  const { data, error } = await query

  if (error) return NextResponse.json({ error: 'Failed to fetch' }, { status: 500 })

  return NextResponse.json({
    data,
    meta: {
      total: data?.length ?? 0,
      last_updated: new Date().toISOString(),
      source: 'LoanMatters — loanmatters.in',
      disclaimer: 'Rates are indicative and sourced from lender websites. Verify with the lender before applying.'
    }
  }, {
    headers: {
      'Cache-Control': 'public, s-maxage=21600, stale-while-revalidate=86400',
      'Access-Control-Allow-Origin': '*'
    }
  })
}

'use client'
import { useState, useMemo } from 'react'

type Filters = {
  country: string
  collateral: 'any' | 'yes' | 'no'
  maxAmountLakhs: number
  lenderType: 'all' | 'psu_bank' | 'private_bank' | 'nbfc'
  sortBy: 'rate' | 'amount' | 'verified'
}

export default function LoanCompareTable({ loans }: { loans: LoanProduct[] }) {
  const [filters, setFilters] = useState<Filters>({
    country: 'any',
    collateral: 'any',
    maxAmountLakhs: 150,
    lenderType: 'all',
    sortBy: 'rate'
  })
  const [selectedIds, setSelectedIds] = useState<string[]>([]) // for side-by-side compare

  const filtered = useMemo(() => {
    return loans
      .filter(l => {
        if (filters.country !== 'any' && !l.countries_covered?.includes(filters.country)) return false
        if (filters.lenderType !== 'all' && l.lenders?.type !== filters.lenderType) return false
        if (filters.collateral === 'no' && l.collateral_required_above_lakhs === null) return false
        return true
      })
      .sort((a, b) => {
        if (filters.sortBy === 'rate') return (a.interest_rate_min ?? 99) - (b.interest_rate_min ?? 99)
        if (filters.sortBy === 'amount') return (b.max_amount_abroad_lakhs ?? 0) - (a.max_amount_abroad_lakhs ?? 0)
        return 0
      })
  }, [loans, filters])

  return (
    <div className="grid grid-cols-1 lg:grid-cols-[280px_1fr] gap-6">
      <FilterSidebar filters={filters} onChange={setFilters} />
      <div className="space-y-4">
        <ResultsHeader count={filtered.length} sortBy={filters.sortBy} onSort={(s) => setFilters(f => ({...f, sortBy: s}))} />
        {filtered.map(loan => (
          <LoanCard
            key={loan.id}
            loan={loan}
            selected={selectedIds.includes(loan.id)}
            onSelect={(id) => setSelectedIds(prev =>
              prev.includes(id) ? prev.filter(x => x !== id) : prev.length < 3 ? [...prev, id] : prev
            )}
          />
        ))}
        {selectedIds.length >= 2 && (
          <StickyCompareBar
            selectedLoans={loans.filter(l => selectedIds.includes(l.id))}
            onClear={() => setSelectedIds([])}
          />
        )}
      </div>
    </div>
  )
}

export async function GET(request: NextRequest) {
  // Verify this is a genuine Vercel cron call
  const authHeader = request.headers.get('authorization')
  if (authHeader !== `Bearer ${process.env.CRON_SECRET}`) {
    return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })
  }

  // For each lender, fetch their official rate page and parse
  // IMPORTANT: Only scrape publicly available rate pages
  // Always show "Rates sourced from [lender] website. Verify before applying."
  const results = await Promise.allSettled([
    updateSBIRates(),
    updateHDFCCredilaRates(),
    updateAvanseRates(),
    // ... other lenders
  ])

  const updated = results.filter(r => r.status === 'fulfilled').length
  const failed = results.filter(r => r.status === 'rejected').length

  // Log to Supabase for admin visibility
  await logCronRun({ updated, failed, ran_at: new Date().toISOString() })

  return NextResponse.json({ updated, failed, timestamp: new Date().toISOString() })
}
